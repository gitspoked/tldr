```python
# tldreadme/peek.py
"""Zero-init codebase reconnaissance — peek at any file or directory."""

import os
from pathlib import Path

SKIP_DIRS = {"node_modules", ".git", "__pycache__", "target", ".venv", "venv", "dist", "build", ".tox"}
MAX_SCAN_DEPTH = 3
MAX_FILE_SIZE = 1_000_000  # 1MB — skip tree-sitter above this


def peek_target(path: str | Path) -> dict:
    """Peek at a file or directory with layered enrichment.

    Layer 0: filesystem scan (always)
    Layer 1: context docs (always)
    Layer 2: indexed knowledge (if .tldr/ exists)
    Layer 3: live services (if responding)
    """
    target = Path(path).resolve()
    enrichment_layers: list[str] = []
    fallback_used: list[str] = []

    if target.is_file():
        result = _scan_file(target)
    else:
        result = _scan_directory(target)

    enrichment_layers.append("disk")

    result.update({
        "path": str(target),
        "enrichment_layers": enrichment_layers,
        "fallback_used": fallback_used,
        "indexed": False,
        "context_docs": [],
        "hot_symbols": [],
        "related": [],
        "generated_summary": None,
    })
    return result


def _scan_directory(target: Path) -> dict:
    """Layer 0: walk the directory, count files/lines by extension."""
    extensions: dict[str, int] = {}
    total_files = 0
    total_lines = 0

    for item in _walk_limited(target, MAX_SCAN_DEPTH):
        if not item.is_file():
            continue
        total_files += 1
        ext = item.suffix.lstrip(".")
        if ext:
            extensions[ext] = extensions.get(ext, 0) + 1
        else:
            extensions["(no ext)"] = extensions.get("(no ext)", 0) + 1
        try:
            total_lines += sum(1 for _ in open(item, "rb"))
        except OSError:
            pass

    return {
        "type": "directory",
        "stats": {
            "files": total_files,
            "lines": total_lines,
            "extensions": dict(sorted(extensions.items(), key=lambda x: -x[1])),
        },
        "symbols": [],
        "project": None,
    }


def _scan_file(target: Path) -> dict:
    """Layer 0: scan a single file for basic stats."""
    try:
        line_count = sum(1 for _ in open(target, "rb"))
    except OSError:
        line_count = 0

    ext = target.suffix.lstrip(".")
    return {
        "type": "file",
        "stats": {
            "files": 1,
            "lines": line_count,
            "extensions": {ext: 1} if ext else {"(no ext)": 1},
        },
        "symbols": [],
        "project": None,
    }


def _walk_limited(root: Path, max_depth: int) -> list[Path]:
    """Walk a directory tree up to max_depth levels deep, skipping noise."""
    results: list[Path] = []

    def _recurse(directory: Path, depth: int):
        if depth > max_depth:
            return
        try:
            entries = sorted(directory.iterdir())
        except OSError:
            return
        for entry in entries:
            if entry.name in SKIP_DIRS or entry.name.startswith("."):
                continue
            if entry.is_symlink():
                continue
            if entry.is_file():
                results.append(entry)
            elif entry.is_dir():
                _recurse(entry, depth + 1)

    _recurse(root, 0)
    return results
```

import re

_SYMBOL_PATTERNS = [
    re.compile(r"^\s*(?:async\s+)?def\s+(\w+)"),         # Python/GDScript
    re.compile(r"^\s*class\s+(\w+)"),                      # Python/Java/TS
    re.compile(r"^\s*(?:pub\s+)?(?:async\s+)?fn\s+(\w+)"), # Rust
    re.compile(r"^\s*func\s+(\w+)"),                       # Go
    re.compile(r"^\s*(?:export\s+)?(?:async\s+)?function\s+(\w+)"),  # JS/TS
    re.compile(r"^\s*(?:pub\s+)?struct\s+(\w+)"),          # Rust
    re.compile(r"^\s*(?:pub\s+)?enum\s+(\w+)"),            # Rust
    re.compile(r"^\s*interface\s+(\w+)"),                   # TS/Java
    re.compile(r"^\s*(?:pub\s+)?trait\s+(\w+)"),           # Rust
]

_KIND_PATTERNS = {
    "class": re.compile(r"^\s*class\s"),
    "struct": re.compile(r"^\s*(?:pub\s+)?struct\s"),
    "enum": re.compile(r"^\s*(?:pub\s+)?enum\s"),
    "interface": re.compile(r"^\s*interface\s"),
    "trait": re.compile(r"^\s*(?:pub\s+)?trait\s"),
}


def _extract_symbols_regex(text: str) -> list[dict]:
    """Extract symbols from source text using regex patterns."""
    symbols: list[dict] = []
    for i, line in enumerate(text.splitlines(), 1):
        for pattern in _SYMBOL_PATTERNS:
            m = pattern.match(line)
            if m:
                name = m.group(1)
                kind = "function"
                for k, kp in _KIND_PATTERNS.items():
                    if kp.match(line):
                        kind = k
                        break
                symbols.append({"name": name, "kind": kind, "line": i})
                break
    return symbols


def _extract_symbols(target: Path) -> list[dict]:
    """Extract symbols from a file — tree-sitter if available, regex fallback."""
    if target.stat().st_size > MAX_FILE_SIZE:
        return []

    # Try tree-sitter first
    try:
        from .asts import parse_file
        result = parse_file(target, isolate=False)
        if result and result.symbols:
            return [
                {"name": s.name, "kind": s.kind, "line": s.line}
                for s in result.symbols
            ]
    except Exception:
        pass

    # Regex fallback
    try:
        text = target.read_text(errors="replace")
    except OSError:
        return []
    return _extract_symbols_regex(text)
```

def _enrich_context_docs(target: Path) -> tuple[list[dict], dict | None]:
    """Layer 1: scan context docs and detect project from manifests."""
    from .context_docs import scan_context_docs
    from .deps import extract_deps_from_directory

    # For files, scan from the parent or project root
    scan_root = target.parent if target.is_file() else target
    # Walk up to find a project root with a manifest
    project_root = _find_project_root(scan_root)
    if project_root != scan_root:
        scan_root = project_root

    raw_docs = scan_context_docs(scan_root)
    context_docs = []
    for doc in raw_docs:
        # Extract first meaningful content as summary
        summary = ""
        for section in doc.sections:
            content = section.get("content", "").strip()
            if content:
                # Take first paragraph, up to 200 chars
                first_para = content.split("\n\n")[0]
                summary = first_para[:200]
                break
        context_docs.append({
            "kind": doc.kind,
            "title": doc.title,
            "summary": summary,
        })

    # Detect project from nearest manifest
    project = None
    deps_list = extract_deps_from_directory(scan_root)
    if deps_list:
        # Pick the manifest nearest to target
        nearest = min(
            deps_list,
            key=lambda d: len(Path(d.manifest_file).parts),
        )
        project = {
            "name": nearest.project_name,
            "version": nearest.project_version,
            "manifest": Path(nearest.manifest_file).name,
        }

    return context_docs, project


def _find_project_root(directory: Path) -> Path:
    """Walk up to find the nearest directory with a project manifest."""
    manifest_names = {"Cargo.toml", "package.json", "go.mod", "pyproject.toml", "setup.py"}
    current = directory
    while current != current.parent:
        if any((current / name).exists() for name in manifest_names):
            return current
        current = current.parent
    return directory
```

def _enrich_live_services(
    target: Path,
    result: dict,
    enrichment_layers: list[str],
    fallback_used: list[str],
):
    """Layer 3: opportunistic enrichment from Qdrant and FalkorDB if running."""
    import os

    qdrant_url = os.getenv("QDRANT_URL", "http://localhost:6333")
    falkordb_url = os.getenv("FALKORDB_URL", "redis://localhost:6379")

    # Try Qdrant
    try:
        import httpx
        resp = httpx.get(f"{qdrant_url}/collections", timeout=1.0)
        if resp.status_code == 200:
            enrichment_layers.append("qdrant")

def render_peek(result: dict) -> str:
    """Render peek result as rich terminal output (plain text with structure)."""
    lines: list[str] = []

    # Header: project name or directory name
    path = Path(result["path"])
    project = result.get("project")
    stats = result["stats"]

    if result["type"] == "directory":
        if project:
            lines.append(f"{project['name']} ({project['manifest']})")
            version_part = f"v{project['version']}" if project["version"] and project["version"] != "0.0.0" else ""
            parts = [p for p in [version_part, f"{stats['files']} files", f"{stats['lines']:,} lines"] if p]
            lines.append(f"   {' . '.join(parts)}")
        else:
            lines.append(f"{path.name}")
            lines.append(f"   {stats['files']} files . {stats['lines']:,} lines")

        # Extension breakdown
        exts = stats.get("extensions", {})
        if exts:
            ext_parts = [f"{k}: {v}" for k, v in list(exts.items())[:8]]
            lines.append(f"   {'  '.join(ext_parts)}")
    else:
        # File mode
        ext = path.suffix.lstrip(".")
        lang = ext.upper() if ext else "Unknown"
        lines.append(f"{path.name} ({lang} . {stats['lines']:,} lines)")
        if project:
            lines.append(f"   Part of: {project['name']} v{project['version']}")

    # What It Is — from context docs
    context_docs = result.get("context_docs", [])
    readme_doc = next((d for d in context_docs if d["kind"] == "readme"), None)
    if readme_doc and readme_doc.get("summary"):
        lines.append("")
        lines.append("-- What It Is " + "-" * 40)
        lines.append(f"{readme_doc['summary']}")

    # Context Docs listing
    other_docs = [d for d in context_docs if d["kind"] != "readme"]
    if other_docs:
        lines.append("")
        lines.append("-- Context Docs " + "-" * 38)
        for doc in other_docs:
            summary_hint = doc.get("summary", "")[:60]
            lines.append(f"{doc['title']:<16} {summary_hint}")

    # Symbols (file mode)
    symbols = result.get("symbols", [])
    if symbols:
        lines.append("")
        lines.append("-- Defines " + "-" * 43)
        for sym in symbols[:15]:
            lines.append(f"{sym['kind']:<10} {sym['name']:<24} line {sym['line']}")

    # Indexed status
    lines.append("")
    lines.append("-- Indexed? " + "-" * 42)
    if result.get("indexed"):
        hot_count = len(result.get("hot_symbols", []))
        lines.append(f".tldr/ exists . hot index: {hot_count} symbols")
        if result.get("generated_summary"):
            lines.append(".claude/TLDR.md generated")
    else:
        lines.append("Not indexed — run `tldr init` for full analysis")

    # Enrichment summary
    layers = result.get("enrichment_layers", [])
    if "qdrant" not in layers:
        lines.append("Qdrant unavailable")
    if "falkordb" not in layers:
        lines.append("FalkorDB unavailable")

    # Related (from Layer 3)
    related = result.get("related", [])
    if related:
        lines.append("")
        lines.append("-- Related " + "-" * 43)
        for r in related[:10]:
            lines.append(f"{r['relationship']}: {r['name']} ({r.get('file', '')})")

    return "\n".join(lines)


def render_peek_markdown(result: dict) -> str:
    """Render peek result as markdown."""
    lines: list[str] = []

    path = Path(result["path"])
    project = result.get("project")
    stats = result["stats"]

    # Title
    name = project["name"] if project else path.name
    lines.append(f"# {name}")

    # Stats
    if result["type"] == "directory":
        if project:
            lines.append(f"**{project['manifest']}** v{project.get('version', '')}")

            ext_parts = [f"`{k}`: {v}" for k, v in list(exts.items())[:8]]
            lines.append(f"Extensions: {', '.join(ext_parts)}")
    else:

        lines.append(f"**{ext.upper()}** file, **{stats['lines']:,}** lines")
        if project:
            lines.append(f"Part of **{project['name']}** v{project.get('version', '')}")

    # Context docs
    context_docs = result.get("context_docs", [])
    if context_docs:
        lines.append("")
        lines.append("## Context")
        for doc in context_docs:
            summary = doc.get("summary", "")
            if summary:
                lines.append(f"- **{doc['title']}** ({doc['kind']}): {summary[:120]}")
            else:
                lines.append(f"- **{doc['title']}** ({doc['kind']})")

    # Symbols
    symbols = result.get("symbols", [])
    if symbols:
        lines.append("")
        lines.append("## Symbols")
        lines.append("")
        lines.append("| Kind | Name | Line |")
        lines.append("|------|------|------|")
        for sym in symbols[:15]:
            lines.append(f"| {sym['kind']} | `{sym['name']}` | {sym['line']} |")

    # Status
    lines.append("")
    lines.append("## Status")
    lines.append(f"- Indexed: {'yes' if result.get('indexed') else 'no'}")
    lines.append(f"- Enrichment: {', '.join(result.get('enrichment_layers', []))}")
    if result.get("fallback_used"):

```python
def peek_to_router_result(peek_result: dict) -> dict:
    """Map a peek result to the router contract shape."""
    layers = peek_result.get("enrichment_layers", [])
    context_docs = peek_result.get("context_docs", [])
    project = peek_result.get("project")
    stats = peek_result.get("stats", {})

    # Summary: prefer README, fall back to project stats
    readme = next((d for d in context_docs if d["kind"] == "readme"), None)
    if readme and readme.get("summary"):
        summary = readme["summary"][:200]
    elif project:
        summary = f"{project['name']}: {stats.get('files', 0)} files, {stats.get('lines', 0):,} lines"
    else:
        target_name = Path(peek_result["path"]).name
        summary = f"{target_name}: {stats.get('files', 0)} files, {stats.get('lines', 0):,} lines"

    # Confidence based on layers present
    if "qdrant" in layers or "falkordb" in layers:
        confidence = 0.9
    elif "tldr" in layers:
        confidence = 0.7
    else:
        confidence = 0.5

    # Evidence
    evidence: list[str] = []
    evidence.append(f"scanned {stats.get('files', 0)} files")
    for doc in context_docs:
        evidence.append(f"found {doc['title']}")
    hot_count = len(peek_result.get("hot_symbols", []))
    if hot_count:
        evidence.append(f"hot index: {hot_count} symbols")
    if peek_result.get("indexed"):
        evidence.append(".tldr/ present")

    # Recommended next action
    if "tldr" not in layers:
        recommended = f"Run `tldr init {peek_result['path']}` for full indexing"
    elif "qdrant" not in layers or "falkordb" not in layers:
        recommended = "Start services (`docker compose up -d`) for richer analysis"
    else:
        recommended = "Codebase fully indexed — use repo_lookup for specific queries"

    return {
        "summary": summary,
        "confidence": confidence,
        "evidence": evidence,
        "recommended_next_action": recommended,
        "fallback_used": peek_result.get("fallback_used", []),
        "peek": peek_result,
    }
```
